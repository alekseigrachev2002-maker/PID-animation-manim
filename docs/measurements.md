## Скетч: лог err по таймеру

Регулятор раз в 5 мс (как в модели), err пишется в память, а не в Serial: печать во время езды сама даёт задержку. Хватает на 2 с езды (Uno: 400 × 2 байта). После прогона моторы стоп, лог печатается в Монитор порта.

```cpp
// вставить вместо followLine() и loop() скетча урока 2

const float speed = 150;
const float sensorBalance = 1.0;
const float kp = 0.3;
const float kd = 0.0;
const uint16_t PERIOD_MS = 5;
const uint16_t N = 400;           // 400 × 5 мс = 2 с

int16_t logErr[N];
uint16_t n = 0;
uint32_t tNext = 0;
float errOld = 0;

void followLineTimed() {
  float err = analogRead(LEFT_SENS) * sensorBalance - analogRead(RIGHT_SENS);
  if (n == 0) errOld = err;       // без рывка D на старте
  float control = err * kp + (err - errOld) * kd;
  motors(speed - control, speed + control);
  errOld = err;
  logErr[n++] = (int16_t)err;
}

void loop() {
  if (tNext == 0) { delay(2000); tNext = millis(); }   // 2 с, чтобы поставить робота
  if (n < N) {
    if (millis() >= tNext) { tNext += PERIOD_MS; followLineTimed(); }
    return;
  }
  motors(0, 0);
  Serial.println("t_ms err");
  for (uint16_t i = 0; i < N; i++) {
    Serial.print(i * PERIOD_MS); Serial.print(' '); Serial.println(logErr[i]);
  }
  while (true) {}
}
```
